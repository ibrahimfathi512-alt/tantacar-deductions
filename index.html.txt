<script>
  // ضع رابط Google Apps Script الذي ينتهي بـ /exec هنا
  const API_URL = "https://script.google.com/macros/s/AKfycbzSE8UbK6-mwSYh5Y3jPLggQeDfcAomFNr5ObGLtKLfZj0fbhK-iXb-SIKtLdUzZYc/exec";

  let currentUser = null;
  let portalState = { isOpen: true, status: 'OPEN', message: '' };
  let driversMap = {};
  let parsedItems = [];
  let historyData = [];

  async function callAPI(data) {
    const response = await fetch(API_URL, {
      method: "POST",
      body: JSON.stringify(data)
    });
    return await response.json();
  }

  window.onload = async function() {
    try {
      const res = await callAPI({ action: 'init' });
      const sel = document.getElementById('zoneSelect');
      sel.innerHTML = '<option value="">-- اختر الزون الخاص بك --</option>';
      (res.zones || []).forEach(z => {
        sel.innerHTML += `<option value="${z}">${z}</option>`;
      });
      sel.innerHTML += '<option value="ADMIN">🔒 دخول الإدارة العامة</option>';
    } catch (e) {
      document.getElementById('zoneSelect').innerHTML = '<option value="ADMIN">🔒 دخول الإدارة العامة</option>';
    }
  };

  async function handleLogin() {
    const zone = document.getElementById('zoneSelect').value;
    const pass = document.getElementById('loginPass').value;
    if (!zone || !pass) {
      document.getElementById('loginError').innerText = 'يرجى اختيار الزون وإدخال الباسورد';
      return;
    }
    const btn = document.getElementById('loginBtn');
    btn.disabled = true;
    btn.innerText = 'جاري الدخول...';

    try {
      const res = await callAPI({ action: 'login', zone, password: pass });
      btn.disabled = false;
      btn.innerText = 'تسجيل الدخول';

      if (res.success) {
        currentUser = res;
        portalState = res.portal;
        document.getElementById('loginScreen').classList.add('d-none');
        document.getElementById('mainApp').classList.remove('d-none');
        document.getElementById('userBadge').innerText = `${res.name} | ${res.zone}`;
        
        if (res.role === 'admin') {
          document.getElementById('tabNewContainer').classList.add('d-none');
          document.getElementById('adminPortalControl').classList.remove('d-none');
          document.querySelectorAll('.admin-col').forEach(el => el.classList.remove('d-none'));
          updateAdminPortalUI();
          switchTab('history');
        } else {
          applySupervisorPortalUI();
          callAPI({ action: 'getDrivers' }).then(map => { driversMap = map; });
          callAPI({ action: 'getHistory', role: res.role, zone: res.zone }).then(data => { historyData = data; });
        }
      } else {
        document.getElementById('loginError').innerText = res.message;
      }
    } catch (e) {
      btn.disabled = false;
      btn.innerText = 'تسجيل الدخول';
      document.getElementById('loginError').innerText = 'تعذر الاتصال بالسيرفر، تأكد من الإنترنت';
    }
  }

  function updateAdminPortalUI() {
    const badge = document.getElementById('adminStatusBadge');
    const btn = document.getElementById('togglePortalBtn');
    document.getElementById('closedMessageInput').value = portalState.message;

    if (portalState.isOpen) {
      badge.className = 'badge bg-success fs-6';
      badge.innerText = '🟢 مفتوح';
      btn.className = 'btn btn-danger btn-sm w-100 fw-bold';
      btn.innerText = '🔒 إغلاق الرفع الآن';
    } else {
      badge.className = 'badge bg-danger fs-6';
      badge.innerText = '🔴 مغلق';
      btn.className = 'btn btn-success btn-sm w-100 fw-bold';
      btn.innerText = '🔓 فتح الرفع للمشرفين';
    }
  }

  async function handleTogglePortal() {
    const nextStatus = portalState.isOpen ? 'CLOSED' : 'OPEN';
    const customMsg = document.getElementById('closedMessageInput').value;
    const btn = document.getElementById('togglePortalBtn');
    btn.disabled = true;
    btn.innerText = 'جاري التحديث...';

    portalState = await callAPI({ action: 'togglePortal', newStatus: nextStatus, customMsg });
    btn.disabled = false;
    updateAdminPortalUI();
  }

  function applySupervisorPortalUI() {
    if (!portalState.isOpen) {
      document.getElementById('openBatchForm').classList.add('d-none');
      document.getElementById('closedAlertBox').classList.remove('d-none');
      document.getElementById('closedAlertText').innerText = portalState.message;
    } else {
      document.getElementById('openBatchForm').classList.remove('d-none');
      document.getElementById('closedAlertBox').classList.add('d-none');
    }
  }

  function switchTab(tab) {
    if (tab === 'new') {
      document.getElementById('newBatchSection').classList.remove('d-none');
      document.getElementById('historySection').classList.add('d-none');
      document.getElementById('newTabBtn').classList.add('active');
      document.getElementById('historyTabBtn').classList.remove('active');
    } else {
      document.getElementById('newBatchSection').classList.add('d-none');
      document.getElementById('historySection').classList.remove('d-none');
      document.getElementById('newTabBtn').classList.remove('active');
      document.getElementById('historyTabBtn').classList.add('active');
      loadHistory();
    }
  }

  function clearInput() {
    document.getElementById('bulkInput').value = '';
    parsedItems = [];
    renderPreview();
  }

  function parseBulkData() {
    const text = document.getElementById('bulkInput').value;
    const defaultReason = document.getElementById('defaultReason').value;
    const lines = text.split('\n');
    parsedItems = [];
    const seenIds = new Set();

    lines.forEach(line => {
      if (!line.trim()) return;
      const nums = line.match(/\d+(\.\d+)?/g);
      if (nums && nums.length >= 1) {
        const id = nums[0];
        const amount = nums.length >= 2 ? parseFloat(nums[1]) : 0;
        let detectedReason = defaultReason;
        if (line.includes('باوتش')) detectedReason = 'باوتشات';
        else if (line.includes('سلف')) detectedReason = 'سلف';
        else if (line.includes('سالب') || line.includes('سوالب')) detectedReason = 'سوالب';

        const driverInfo = driversMap[id] || { name: 'غير مسجل' };
        parsedItems.push({
          id, driverName: driverInfo.name, amount, reason: detectedReason, isDuplicate: seenIds.has(id)
        });
        seenIds.add(id);
      }
    });
    renderPreview();
  }

  function renderPreview() {
    const tbody = document.getElementById('previewTableBody');
    tbody.innerHTML = '';

    if (parsedItems.length === 0) {
      tbody.innerHTML = '<tr><td colspan="5" class="text-muted py-3">البيانات التي ستلصقها بالأعلى ستظهر هنا تلقائياً...</td></tr>';
      document.getElementById('totalCount').innerText = '0';
      document.getElementById('totalAmount').innerText = '0 ج.م';
      return;
    }

    let total = 0;
    parsedItems.forEach((item, index) => {
      total += (item.amount || 0);
      const rowClass = item.isDuplicate ? 'table-warning' : (item.amount <= 0 ? 'table-danger' : '');
      tbody.innerHTML += `
        <tr class="${rowClass}">
          <td class="fw-bold">${item.id} ${item.isDuplicate ? '<span class="badge bg-danger">مكرر</span>' : ''}</td>
          <td>${item.driverName}</td>
          <td><input type="number" class="form-control form-control-sm text-center fw-bold" value="${item.amount || ''}" onchange="updateItem(${index}, 'amount', this.value)"></td>
          <td>
            <select class="form-select form-select-sm" onchange="updateItem(${index}, 'reason', this.value)">
              <option value="باوتشات" ${item.reason === 'باوتشات' ? 'selected' : ''}>باوتشات</option>
              <option value="سلف" ${item.reason === 'سلف' ? 'selected' : ''}>سلف</option>
              <option value="سوالب" ${item.reason === 'سوالب' ? 'selected' : ''}>سوالب</option>
            </select>
          </td>
          <td><button class="btn btn-sm btn-outline-danger" onclick="removeItem(${index})">✕</button></td>
        </tr>`;
    });

    document.getElementById('totalCount').innerText = parsedItems.length;
    document.getElementById('totalAmount').innerText = total.toLocaleString() + ' ج.م';
  }

  function updateItem(index, field, val) {
    parsedItems[index][field] = field === 'amount' ? parseFloat(val) || 0 : val;
    renderPreview();
  }

  function removeItem(index) {
    parsedItems.splice(index, 1);
    renderPreview();
  }

  async function submitData() {
    if (parsedItems.length === 0) return alert('يرجى لصق بيانات الطيارين أولاً!');
    if (parsedItems.some(i => !i.amount || i.amount <= 0)) return alert('يرجى كتابة المبلغ لجميع الطيارين.');

    const btn = document.getElementById('submitBtn');
    btn.disabled = true;
    btn.innerText = '⚡ جاري الحفظ...';

    const payload = { supervisorName: currentUser.name, zone: currentUser.zone, items: [...parsedItems] };
    const res = await callAPI({ action: 'submitBatch', payload });

    btn.disabled = false;
    btn.innerText = '🚀 اعتماد ورفع الدفعة للإدارة';

    if (res.success) {
      const today = res.timestamp.split(' ')[0];
      payload.items.forEach(item => {
        historyData.unshift({
          batchId: res.batchId, date: today, supervisor: currentUser.name, zone: currentUser.zone,
          driverId: item.id, driverName: item.driverName, amount: item.amount,
          reason: item.reason, status: 'قيد المراجعة', note: ''
        });
      });
      clearInput();
      document.getElementById('newBatchSection').classList.add('d-none');
      document.getElementById('historySection').classList.remove('d-none');
      document.getElementById('newTabBtn').classList.remove('active');
      document.getElementById('historyTabBtn').classList.add('active');
      renderHistory();
    } else {
      alert(res.message);
      location.reload();
    }
  }

  async function loadHistory() {
    const tbody = document.getElementById('historyTableBody');
    tbody.innerHTML = '<tr><td colspan="9" class="py-4">جاري تحميل السجل...</td></tr>';
    historyData = await callAPI({ action: 'getHistory', role: currentUser.role, zone: currentUser.zone });
    renderHistory();
  }

  function renderHistory() {
    const search = document.getElementById('searchInput').value.toLowerCase();
    const statusFilter = document.getElementById('filterStatus').value;
    const reasonFilter = document.getElementById('filterReason').value;
    const tbody = document.getElementById('historyTableBody');
    tbody.innerHTML = '';

    let total = 0, uploaded = 0, rejected = 0;

    historyData.filter(r => 
      (r.driverId.includes(search) || r.driverName.toLowerCase().includes(search)) &&
      (statusFilter === 'ALL' || r.status === statusFilter) &&
      (reasonFilter === 'ALL' || r.reason === reasonFilter)
    ).forEach(r => {
      total += r.amount;
      if (r.status === 'تم الرفع') uploaded += r.amount;
      if (r.status === 'لم يرفع') rejected += r.amount;

      let badgeClass = r.status === 'تم الرفع' ? 'badge-uploaded' : (r.status === 'لم يرفع' ? 'badge-rejected' : 'badge-pending');
      const adminControls = currentUser.role === 'admin' ? `
        <td>
          <div class="btn-group btn-group-sm">
            <button class="btn btn-success" onclick="changeStatus(${r.archRowIndex}, '${r.uid}', 'تم الرفع')">✓</button>
            <button class="btn btn-danger" onclick="rejectPrompt(${r.archRowIndex}, '${r.uid}')">✕</button>
          </div>
        </td>` : '';
      const adminSupCol = currentUser.role === 'admin' ? `<td><strong>${r.zone}</strong><br><small>${r.supervisor}</small></td>` : '';

      tbody.innerHTML += `
        <tr>
          <td>${r.date.split(' ')[0]}</td>
          ${adminSupCol}
          <td class="fw-bold">${r.driverId}</td>
          <td>${r.driverName}</td>
          <td class="fw-bold">${r.amount}</td>
          <td>${r.reason}</td>
          <td><span class="badge ${badgeClass}">${r.status}</span></td>
          <td class="text-danger">${r.note || '-'}</td>
          ${adminControls}
        </tr>`;
    });

    document.getElementById('histTotal').innerText = total.toLocaleString();
    document.getElementById('histUploaded').innerText = uploaded.toLocaleString();
    document.getElementById('histRejected').innerText = rejected.toLocaleString();
  }

  async function changeStatus(archRowIndex, uid, newStatus, adminNote = '') {
    await callAPI({ action: 'updateStatus', archRowIndex, uid, newStatus, adminNote });
    loadHistory();
  }
